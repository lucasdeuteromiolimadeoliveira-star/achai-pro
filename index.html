O usuário busca: "${query}". Responda de forma curta e persuasiva em HTML simples (use apenas
    ,

Ao final, forneça apenas um termo de busca curto (máximo 3 palavras) dentro de uma tag oculta: TERMO AQUI`; try { const response = await fetch(`https://googleapis.com{apiKey}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] }) }); const data = await response.json(); if (!response.ok) { return res.status(500).json({ error: data.error?.message || 'Erro na API do Gemini' }); } const aiText = data.candidates[0].content.parts[0].text; return res.status(200).json({ resposta: aiText }); } catch (error) { return res.status(500).json({ error: 'Falha ao consultar a IA.' }); } }
